-- Creating first keyspace
CREATE KEYSPACE IF NOT EXISTS my_first_keyspace WITH replication = {'class': 'SimpleStrategy', 'replication_factor' : 3};

-- Using first keyspace
USE my_first_keyspace;

-- List existing keyspaces
DESC KEYSPACES;


CREATE KEYSPACE my_second_keyspace WITH replication = {'class':'SimpleStrategy', 'replication_factor': 1};
DESC KEYSPACES;

/*
my_first_keyspace   system_auth         system_traces        
my_second_keyspace  system_distributed  system_views         
system              system_schema       system_virtual_schema
*/

-- Command to drop keyspace

DROP KEYSPACE "my_second_keyspace";
DESC KEYSPACES;

/*
my_first_keyspace  system_auth         system_schema  system_views         
system             system_distributed  system_traces  system_virtual_schema
*/

-- Command to alter keyspace

ALTER KEYSPACE my_first_keyspace WITH replication={'class': 'SimpleStrategy', 'replication_factor': 2};

-- Command to create table with partition key groupname and clustering key username
CREATE TABLE groups(
                     ... groupname text,
                     ... username text,
                     ... email text,
                     ... age int,
                     ... PRIMARY KEY (groupname, username)
                     ... );

-- Command to insert into table

INSERT INTO groups(groupname, username, email,age) VALUES ('A', 'andre', 'andrebutuc@ua.pt', 20);
INSERT INTO groups(groupname, username, email,age) VALUES ('A', 'bruno', 'bruno@ua.pt', 19);

INSERT INTO groups(groupname, username, email,age) VALUES ('C', 'david', 'david@ua.pt', 22);


-- Command to select all columns

-- Using select with specific return columns
SELECT username,age FROM groups ;

/*
 username | age
----------+-----
    david |  22
   carlos |  21
    andre |  20
    bruno |  19
*/

-- Using select with WHERE <partition_key> '=' value
SELECT username,age FROM groups WHERE groupname='A';

/*
 username | age
----------+-----
    andre |  20
    bruno |  19
*/

-- If we try the previous statement with other keys beside the partition key an error occurs.
SELECT username,age FROM groups WHERE username='andre';

/*
InvalidRequest: Error from server: code=2200 [Invalid query] message="Cannot execute this query as it might involve data filtering and thus may have unpredictable performance. If you want to execute this query despite the performance unpredictability, use ALLOW FILTERING"
*/

-- If we want to force such selection criteria, we must add the flag ALLOW FILTERING

SELECT username,age FROM groups WHERE username='andre' ALLOW FILTERING;

/*
 username | age
----------+-----
    andre |  20
*/

-- The same error happens if we add an AND operator and excluding the clustering key:
SELECT username,age FROM groups WHERE groupname = 'A' AND age=19;

/*
InvalidRequest: Error from server: code=2200 [Invalid query] message="Cannot execute this query as it might involve data filtering and thus may have unpredictable performance. If you want to execute this query despite the performance unpredictability, use ALLOW FILTERING"
*/


-- The error is solved by adding ALLOW FILTERING
SELECT username,age FROM groups WHERE groupname = 'A' AND age=19 ALLOW FILTERING;

/*
 username | age
----------+-----
    bruno |  19
*/

-- If we want to run it without the ALLOW FILTERING, we must follow the key hierarchy and only test the value of the clustering key
SELECT username,age FROM groups WHERE groupname = 'A' AND username='bruno';

/*
 username | age
----------+-----
    bruno |  19
*/

-- If we really want to select rows without having to add the ALLOW FILTERING we must create an index on that attribute field
CREATE INDEX ON groups(age);
