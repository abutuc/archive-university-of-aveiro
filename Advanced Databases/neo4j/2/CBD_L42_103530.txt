#3

return a, relations, b

#4

#5
match (p1:Person)-[:ACTED_IN]->(m:Movie)<-[:ACTED_IN]-(p2:Person) 
with p1, count(m) as movies, p2 
where movies > 1 and id(p1) > id(p2) 
return p1.name, p2.name, movies

#6 // Average: 40.2
match (p:Person)-[:ACTED_IN]->(m:Movie {title: "Apollo 13"}) 
with avg(m.released - p.born) as avg_age, m 
return m.title, avg_age

#7
match (p:Person)-[:ACTED_IN]->(m:Movie) 
with avg(m.released - p.born) as avg_age, m 
return m.title, round(avg_age, 2) as avg_age 
order by avg_age desc
limit 10

#8 // Stand By Me
match (p:Person)-[acted:ACTED_IN]->(m:Movie) 
with avg(m.released - p.born) as avg_age, m, collect(p) as people, collect(acted) as acted 
return people, acted, m 
order by avg_age asc 
limit 1

// Apresente o top 10 dos atores mais velhos (usar o ano de 2022) ordenados pelo número de filmes em que atuaram
match (p:Person)-[:ACTED_IN]->(:Movie) 
with p, count(*) as acted, 2022-p.born as age 
where age > 0 
return age, acted, p.name 
order by age desc, acted desc limit 10

// Apresentar o top 10 dos filmes melhor cotados (rating arredondado às décimas) 
match (reviewer:Person)-[review:REVIEWED]->(m:Movie) 
with round(avg(review.rating),1) as avg_rating, m 
return m.title, avg_rating 
order by avg_rating desc 
limit 10
